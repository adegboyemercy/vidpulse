/**
 * Resilient multi-model Gemini caller with automatic fallback:
 * Tries 'gemini-3.8-flash' first, then 'gemini-3.1-flash-lite' if busy (503) or unavailable.
 */
async function generateWithGemini(
  prompt: string,
  isJson = true,
  temperature = 0.7
): Promise<any> {
  const models = ['gemini-3.8-flash', 'gemini-3.1-flash-lite'];
  let lastError: any = null;

  for (const model of models) {
    try {
      const response = await ai.models.generateContent({
        model,
        contents: prompt,
        config: isJson
          ? { responseMimeType: 'application/json', temperature }
          : { temperature },
      });

      const text = response.text?.trim();
      if (text) {
        if (isJson) {
          // Clean possible markdown code fence wrapping ```json ... ```
          const cleaned = text.replace(/^```json\s*/i, '').replace(/\s*```$/, '').trim();
          return JSON.parse(cleaned);
        }
        return text;
      }
    } catch (err: any) {
      console.warn(`Model ${model} request failed: ${err?.message || err}. Trying next model...`);
      lastError = err;
    }
  }

  throw lastError || new Error('All Gemini text models were unavailable.');
}

/**
 * Converts 16-bit PCM bytes (24kHz mono) into a valid WAV Buffer with standard 44-byte RIFF header
 */
function pcmToWavBuffer(
  pcmBytes: Uint8Array,
  sampleRate = 24000,
  numChannels = 1,
  bitsPerSample = 16
): Buffer {
  const byteRate = (sampleRate * numChannels * bitsPerSample) / 8;
  const blockAlign = (numChannels * bitsPerSample) / 8;
  const totalDataLen = pcmBytes.length;
  const totalFileLen = 36 + totalDataLen;

  const header = Buffer.alloc(44);
  header.write('RIFF', 0);
  header.writeUInt32LE(totalFileLen, 4);
  header.write('WAVE', 8);
  header.write('fmt ', 12);
  header.writeUInt32LE(16, 16);
  header.writeUInt16LE(1, 20);
  header.writeUInt16LE(numChannels, 22);
  header.writeUInt32LE(sampleRate, 24);
  header.writeUInt32LE(byteRate, 28);
  header.writeUInt16LE(blockAlign, 32);
  header.writeUInt16LE(bitsPerSample, 34);
  header.write('data', 36);
  header.writeUInt32LE(totalDataLen, 40);

  return Buffer.concat([header, Buffer.from(pcmBytes)]);
}

/**
 * String hash helper for deterministic variance
 */
function hashString(str: string): number {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = (hash << 5) - hash + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash);
}

function formatNumber(num: number): string {
  if (!num || isNaN(num)) return '0';
  if (num >= 1000000000) return (num / 1000000000).toFixed(1).replace(/\.0$/, '') + 'B';
  if (num >= 1000000) return (num / 1000000).toFixed(1).replace(/\.0$/, '') + 'M';
  if (num >= 1000) return (num / 1000).toFixed(1).replace(/\.0$/, '') + 'K';
  return num.toLocaleString();
}

function parseSubscribersString(str: string): number {
  if (!str) return 0;
  const clean = str.toLowerCase().replace(/,/g, '').trim();
  const numMatch = clean.match(/([0-9.]+)\s*([kmb])?/);
  if (!numMatch) return 0;
  const val = parseFloat(numMatch[1]);
  const unit = numMatch[2];
  if (unit === 'm') return Math.round(val * 1000000);
  if (unit === 'k') return Math.round(val * 1000);
  if (unit === 'b') return Math.round(val * 1000000000);
  return Math.round(val);
}

function extractVideoId(input: string): string | null {
  const cleanInput = input.trim();
  const patterns = [
    /(?:youtu\.be\/|v\/|u\/\w\/|embed\/|watch\?v=|shorts\/|live\/)([\w-]{11})/,
    /^([\w-]{11})$/,
  ];
  for (const pattern of patterns) {
    const match = cleanInput.match(pattern);
    if (match && match[1]) {
      return match[1];
    }
  }
  return null;
}

function extractHandleOrName(input: string): string {
  let clean = input.trim();
  // Remove URL protocol and domain
  clean = clean.replace(/^https?:\/\/(?:www\.)?youtube\.com\//i, '');
  clean = clean.replace(/^https?:\/\/(?:www\.)?youtu\.be\//i, '');
  // Remove trailing path segments like /videos, /featured, /about
  clean = clean.replace(/\/(?:videos|shorts|playlists|community|featured|about|streams).*$/i, '');
  // Remove query params
  clean = clean.replace(/\?.*$/, '');
  // Remove leading @ or /@ or /c/ or /user/
  clean = clean.replace(/^(?:@|\/?@|\/?c\/|\/?user\/|\/?channel\/)/i, '');
  return clean.trim();
}

  // 3. Text to Speech Endpoint (MANDATORY: gemini-3.8-flash-tts)
  app.post('/api/gemini/tts', async (req: Request, res: Response) => {
    try {
      const { text, voiceName = 'Puck', style } = req.body;

      if (!text || typeof text !== 'string' || text.trim().length === 0) {
        res.status(400).json({ error: 'Text prompt is required for TTS synthesis.' });
        return;
      }

      const styleInstruction = style || 'Enthusiastic, energetic, clear, high-retention YouTube creator voice';
      const selectedVoice = ['Puck', 'Charon', 'Kore', 'Fenrir', 'Zephyr'].includes(voiceName) ? voiceName : 'Puck';

      // Call gemini-3.8-flash-tts
      const response = await ai.models.generateContent({
        model: 'gemini-3.8-flash-tts',
        contents: [
          {
            role: 'user',
            parts: [
              {
                text: text.trim(),
                speechMetadata: {
                  speaker: 'Creator',
                  style: styleInstruction,
                },
              },
            ],
          },
        ],
        config: {
          responseModalities: ['AUDIO'],
          speechConfig: {
            voiceConfig: {
              prebuiltVoiceConfig: { voiceName: selectedVoice },
            },
          },
        },
      });

      const base64Audio = response.candidates?.[0]?.content?.parts?.[0]?.inlineData?.data;
      if (!base64Audio) {
        res.status(500).json({ error: 'TTS audio data was not returned by Gemini.' });
        return;
      }

      // Convert raw 16-bit PCM bytes (24,000 Hz, 1 channel) into WAV with RIFF header
      const pcmBytes = Buffer.from(base64Audio, 'base64');
      const wavBuffer = pcmToWavBuffer(pcmBytes, 24000, 1, 16);
      const wavBase64 = wavBuffer.toString('base64');
      const audioDataUrl = `data:audio/wav;base64,${wavBase64}`;

      const durationSeconds = +(pcmBytes.length / (24000 * 2)).toFixed(1);

      res.json({
        success: true,
        model: 'gemini-3.8-flash-tts',
        voice: selectedVoice,
        audioUrl: audioDataUrl,
        durationSeconds,
      });
    } catch (error: any) {
      console.error('Error generating TTS:', error);
      res.status(500).json({ error: error.message || 'Failed to synthesize speech.' });
    }
  });
